  // ----------------------------------------------------------------------
  // syntactic domains
  // ----------------------------------------------------------------------
  domains
  {
    Program = Prog[Sequence];
    Sequence = Seq[Sequence,Sequence] + Cons[Constant] + ArithmeticOp[ArithmeticOperations] + LogicalOp[LogicalOperations] + StackOp[StackOperations] + FunctionOp[FunctionOperations] + ConditionsOp[ConditionsOperation];
    Constant = Num[NUM] + True + False;
    ArithmeticOperations = Add + Sub + Mul + Cmp;
    LogicalOperations = Not + And + IsNeg + IsPos;
    StackOperations = Clear + Id + Pop + Dup + Over + Swap + Rotl;
    FunctionOperations = Compose + Apply + ApplyOver + Quote[Sequence];
    ConditionsOperation = Choose[Sequence,Sequence,Sequence] + While;
  }

  // -----------------------------------------------------------------------
  // printer
  // -----------------------------------------------------------------------
  printer
  {
    domain Program
    {
      case Prog[s] → # _result = s.toString(); #;
    }
    domain Sequence
    {
      case Seq[s1,s2] → # _result =  s1 + " " + s2; #;
      case Cons[c] → # _result = c.toString(); #;
      case ArithmeticOp[a] → # _result = a.toString(); #;
      case LogicalOp[l] → # _result = l.toString(); #;
      case StackOp[s] → # _result = s.toString(); #;
      case FunctionOp[f] → # _result = f.toString(); #;
      case ConditionsOp[o] → # _result = o.toString(); #;
    }
    domain Constant
    {
      case Num[n] → # _result = n; #;
      case True → # _result = "TRUE"; #;
      case False → # _result = "FALSE"; #;
    }
    domain ArithmeticOperations
    {
      case Add → # _result = "ADD"; #;
      case Sub → # _result = "SUB"; #;
      case Mul → # _result = "MUL"; #;
      case Cmp → # _result = "CMP"; #;
    }
    domain LogicalOperations
    {
      case Not → # _result = "NOT"; #;
      case And → # _result = "AND"; #;
      case IsNeg → # _result = "ISNEG"; #;
      case IsPos → # _result = "ISPOS"; #;
    }
    domain StackOperations
    {
      case Clear → # _result = "\n"; #;
      case Id → # _result = "ID"; #;
      case Pop → # _result = "POP"; #;
      case Dup → # _result = "DUP"; #;
      case Over → # _result = "OVER"; #;
      case Swap → # _result = "SWAP"; #;
      case Rotl → # _result = "ROTL"; #;
    }
    domain FunctionOperations
    {
      case Compose → # _result = "COMPOSE"; #;
      case Apply → # _result = "APPLY"; #;
      case ApplyOver → # _result = "APPLYOVER"; #;
      case Quote[s] → # _result = "{" + s + "}"; #;
    }
    domain ConditionsOperation
    {
      case Choose[s1,s2,s3] → # _result = "(" + s1 + ") CHOOSE (" + s2 + ") (" + s3 + ")"; #; //TODO: try add option for condition without parentheses
      case While → # _result = "WHILE"; #;
    }
  }

  // -----------------------------------------------------------------------
  // parser
  // -----------------------------------------------------------------------
  parser antlr4
  {
    domain Program
    {
      case # s=dSequence EOF # → Prog[s];
    }
    domain Sequence
    {
      case # s1=dSequence ' ' s2=dSequence # → Seq[s1,s2];
      case # c=dConstant # → Cons[c];
      case # a=dArithmeticOperations # → ArithmeticOp[a];
      case # l=dLogicalOperations # → LogicalOp[l];
      case # s=dStackOperations # → StackOp[s];
      case # f=dFunctionOperations # → FunctionOp[f];
      case # o=dConditionsOperation # → ConditionsOp[o];
    }
    domain Constant
    {
      case # n=dNUM # → Num[n];
      case # 'TRUE' # → True;
      case # 'FALSE' # → False;
    }
    domain ArithmeticOperations
    {
      case # 'ADD' # → Add;
      case # 'SUB' # → Sub;
      case # 'MUL' # → Mul;
      case # 'CMP' # → Cmp;
    }
    domain LogicalOperations
    {
      case # 'NOT'# → Not;
      case # 'AND' # → And;
      case # 'ISNEG'# → IsNeg;
      case # 'ISPOS'# → IsPos;
    }
    domain StackOperations
    {
      case # 'CLEAR'# → Clear;
      case # 'ID' # → Id;
      case # 'POP'# → Pop;
      case # 'DUP'# → Dup;
      case # 'OVER' # → Over;
      case # 'SWAP'# → Swap;
      case # 'ROTL'# → Rotl;
    }
    domain FunctionOperations
    {
      case # 'COMPOSE'# → Compose;
      case # 'APPLY' # → Apply;
      case # 'APPLYOVER'# → ApplyOver;
      case # '{ ' + s=dSequence  + ' }'# → Quote[s];
    }
    domain ConditionsOperation
    {
      case # '( ' + s1=dSequence  + ' ) CHOOSE ( ' + s2=dSequence  + ' ) ( ' + s3=dSequence  + ' )'# → Choose[s1,s2,s3];
      case # 'WHILE' # → While;
    }
  }
