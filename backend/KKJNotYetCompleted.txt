  // ----------------------------------------------------------------------
  // syntactic domains
  // ----------------------------------------------------------------------
  domains
  {
    NumberConstant = Num[NUM];
    BooleanConstant = True + False;
    Constant = NumConstant[NumberConstant] + BoolConstant[BooleanConstant] + NumPopInside[NumberConstant,Constant] + BoolPopInside[BooleanConstant,Constant];
    NumberExpression = NumExpression[NumberConstant] + Add[NumberExpression,NumberExpression] + Sub[NumberExpression,NumberExpression] + Mul[NumberExpression,NumberExpression] + Cmp[NumberExpression,NumberExpression] + NumberPopBefore[NumberExpression,Constant] + NumberPopAfter[Constant,NumberExpression] + NumberPopInside1[Constant,NumberExpression] + NumberPopInside2[Constant,NumberExpression];
    BooleanExpression = BoolExpression[BooleanConstant] + Not[BooleanExpression] + And[BooleanExpression,BooleanExpression] + IsNeg[NumberExpression] + IsPos[NumberExpression] + BooleanPopBefore[BooleanExpression,Constant] + BooleanPopAfter[Constant,BooleanExpression] + BooleanPopInside1[Constant,BooleanExpression] + BooleanPopInside2[Constant,BooleanExpression];
    Command = Exp[NumberExpression] + BoolExp[BooleanExpression] + Pop[Constant];
    Sequence = Single[Command] + Seq[Command,Sequence];
    Program = Prog[Sequence];
  }

  // -----------------------------------------------------------------------
  // printer

  printer
  {
    domain NumberConstant
    {
      case Num[n] → # _result = n; #;
    }
    domain BooleanConstant
    {
      case True → # _result = "true"; #;
      case False → # _result = "false"; #;
    }
    domain Constant
    {
      case NumConstant[n] → # _result = n.toString(); #;
      case BoolConstant[b] → # _result = b.toString(); #;
      case NumPopInside[n,c] → # _result = " X" + n + "X " + c; #;
      case BoolPopInside[b,c] → # _result = " X" + b + "X " + c; #;
    }
    domain NumberExpression
    {
      case NumExpression[n] → # _result = n.toString(); #;
      case Add[e1,e2] → # _result = "(" + e1 + " + " + e2 + ")"; #;
      case Sub[e1,e2] → # _result = "(" + e1 + " - " + e2 + ")"; #;
      case Mul[e1,e2] → # _result = "(" + e1 + " * " + e2 + ")"; #;
      case Cmp[e1,e2] → # _result = "(" + e1 + " == " + e2 + ")"; #;
      case NumberPopBefore[e,c] → # _result = e + " X" + c + "X "; #;
      case NumberPopAfter[c,e] → # _result = " X" + c + "X " + e; #;
      case NumberPopInside1[c,e] → # _result = " X" + c + "X " + e; #;
      case NumberPopInside2[c,e] → # _result = " X" + c + "X " + " X" + e + "X "; #;
    }
    domain BooleanExpression
    {
      case BoolExpression[bc] → # _result = bc.toString(); #;
      case Not[b] → # _result = "(~" + b + ")"; #;
      case And[b1,b2] → # _result = "(" + b1 + " && " + b2 + ")"; #;
      case IsNeg[e] → # _result = "(" + e + " < 0 )"; #;
      case IsPos[e] → # _result = "(" + e + " > 0 )"; #;
      case BooleanPopBefore[b,c] → # _result = b + " X" + c + "X "; #;
      case BooleanPopAfter[c,b] → # _result = " X" + c + "X " + b; #;
      case BooleanPopInside1[c,b] → # _result = " X" + c + "X " + b; #;
      case BooleanPopInside2[c,b] → # _result = " X" + c + "X " + " X" + b + "X "; #;
    }
    domain Command
    {
      case Exp[e] → # _result = e.toString(); #;
      case BoolExp[b] → # _result = b.toString(); #;
      case Pop[c] → # _result = " X" + c + "X "; #;
    }
    domain Sequence
    {
      case Single[c] → # _result = "{" + c + "}"; #;
      case Seq[c1,c2] → # _result = "{" + c1 + "}" + ", " + c2; #;
    }
    domain Program
    {
      case Prog[c] → # _result = c.toString(); #;
    }
  }
  
  // -----------------------------------------------------------------------

  // -----------------------------------------------------------------------
  parser antlr4
  {
    domain NumberConstant
    {
      case # n=dNUM # → Num[n];
    }
    domain BooleanConstant
    {
      case # 'true' # → True;
      case # 'false' # → False;
    }
    domain Constant
    {
      case # n=dNumberConstant # → NumConstant[n];
      case # b=dBooleanConstant # → BoolConstant[b];
      case # n=dNumberConstant ' ' c=dConstant ' POP' # → NumPopInside[n,c];
      case # b=dBooleanConstant ' ' c=dConstant ' POP' # → BoolPopInside[b,c];
    }
    domain NumberExpression
    {
      case # n=dNumberConstant # → NumExpression[n];
      case # e1=dNumberExpression ' ' e2=dNumberExpression ' MUL' # → Mul[e1,e2];
      case # e1=dNumberExpression ' ' e2=dNumberExpression ' ADD' # → Add[e1,e2];
      case # e1=dNumberExpression ' ' e2=dNumberExpression ' SUB' # → Sub[e1,e2];
      case # e1=dNumberExpression ' ' e2=dNumberExpression ' CMP' # → Cmp[e1,e2];
      case # e=dNumberExpression ' ' c=dConstant ' POP' # → NumberPopBefore[e,c];
      case # c=dConstant ' POP' ' ' e=dNumberExpression # → NumberPopAfter[c,e];
      case # c=dConstant ' ' e=dNumberExpression ' POP' # → NumberPopInside1[c,e];
      case # c=dConstant ' ' e=dNumberExpression ' POP POP' # → NumberPopInside2[c,e];
    }
    domain BooleanExpression
    {
      case # bc=dBooleanConstant # → BoolExpression[bc];
      case # b=dBooleanExpression ' NOT'# → Not[b];
      case # b1=dBooleanExpression ' ' b2=dBooleanExpression ' AND' # → And[b1,b2];
      case # e=dNumberExpression ' ISNEG'# → IsNeg[e];
      case # e=dNumberExpression ' ISPOS'# → IsPos[e];
      case # b=dBooleanExpression ' ' c=dConstant ' POP' # → BooleanPopBefore[b,c];
      case # c=dConstant ' POP' ' ' b=dBooleanExpression # → BooleanPopAfter[c,b];
      case # c=dConstant ' ' b=dBooleanExpression ' POP' # → BooleanPopInside1[c,b];
      case # c=dConstant ' ' b=dBooleanExpression ' POP POP' # → BooleanPopInside2[c,b];
    }
    domain Command
    {
      case # e=dNumberExpression # → Exp[e];
      case # b=dBooleanExpression # → BoolExp[b];
      case # c=dConstant ' POP' # → Pop[c];
    }
    domain Sequence
    {
      case # c=dCommand # → Single[c];
      case # c1=dCommand ' ' c2=dSequence # → Seq[c1,c2];
    }
    domain Program
    {
      case # c=dSequence EOF # → Prog[c];
    }
  }

  {#
    public enum Type { Int }
    public static class Env
    {
      private HashMap<String,Value> variables = new HashMap<>();
      private List<Stack<Value>> stacks = new LinkedList<>();
      public Env()
      { super(); stacks.add(new Stack<>()); }
      public void put(String key, Value value)
      { variables.put(key.toString(), value); }
      public Value get(String key)
      { return variables.get(key.toString()); }
      public String printVariables()
      { return variables.toString(); }
      public void push(Value value)
      { stacks.getLast().push(value); }
      public Value peek()
      { if (!stacks.getLast().isEmpty()) { return stacks.getLast().peek(); } else { throw new IllegalStateException("Cannot peek from an empty stack."); } }
      public Value pop()
      { if (!stacks.getLast().isEmpty()) { return stacks.getLast().pop(); } else { throw new IllegalStateException("Cannot pop from an empty stack."); } }
      public String printStack()
      { return stacks.getLast().toString(); }
      public  String printStackHistory()
      { return stacks.toString(); }
      public void newStack()
      { stacks.add(new Stack<>()); }
    }
    public static class TypeError extends RuntimeException
    { public TypeError(String msg) { super(msg); } }
    public static void check(Boolean b, String msg)
    { if (!b) throw new TypeError(msg); }
    sealed interface Value permits IntegerValue, StringValue, BoolValue { Object getValue(); }
    record IntegerValue(Integer value) implements Value
    { @Override public Integer getValue() { return value; } @Override public String toString() { return String.valueOf(value); } }
    record StringValue(String value) implements Value
    { @Override public String getValue() { return value; } @Override public String toString() { return value; } }
    record BoolValue(Boolean value) implements Value
    { @Override public Boolean getValue() { return value; } @Override public String toString() { return String.valueOf(value); } }
  #}
